<script>
// In-memory demo users
let users = [
  {name:"Admin", email:"admin@gmail.com", password:"admin123", role:"admin"},
  {name:"Doctor", email:"doctor@gmail.com", password:"doc123", role:"doctor"},
  {name:"Nurse", email:"nurse@gmail.com", password:"nurse123", role:"nurse"},
  {name:"Patient", email:"patient@gmail.com", password:"patient123", role:"patient"}
];

let tempUser = null;
let currentOTP = null;

function showForm(id){
  document.querySelectorAll('.auth-section').forEach(s=>s.style.display='none');
  document.getElementById(id).style.display='block';
}

// REGISTER
function register(){
  const name = document.getElementById("registerName").value.trim();
  const email = document.getElementById("registerEmail").value.trim();
  const password = document.getElementById("registerPassword").value;
  const confirm = document.getElementById("registerConfirmPassword").value;
  const role = document.getElementById("registerRole").value;

  if(!name || !email || !password || !confirm || !role){
    alert("Fill all fields!");
    return;
  }
  if(password !== confirm){
    alert("Passwords do not match!");
    return;
  }
  if(users.find(u=>u.email===email)){
    alert("Email already registered!");
    return;
  }
  users.push({name,email,password,role});
  alert("Registration successful! Login now.");
  showForm('login-section');
}

// LOGIN
function login(){
  const email = document.getElementById("loginEmail").value.trim();
  const password = document.getElementById("loginPassword").value;
  const user = users.find(u=>u.email===email && u.password===password);
  if(!user){ alert("Invalid credentials"); return; }
  tempUser = user;
  generateOTP();
  showForm('twofa-section');
}

// GENERATE DEMO OTP
function generateOTP(){
  currentOTP = Math.floor(100000 + Math.random()*900000);
  document.getElementById("demoOTP").innerText = currentOTP;
}

// VERIFY 2FA
function verify2FA(){
  const code = document.getElementById("twofaCode").value;
  if(code == currentOTP){
    alert(`Login successful! Welcome ${tempUser.name} (${tempUser.role})`);
    // Redirect based on role
    if(tempUser.role === "admin") window.location.href="admin-dashboard.html";
    else if(tempUser.role === "doctor") window.location.href="doctor-dashboard.html";
    else if(tempUser.role === "nurse") window.location.href="nurse-dashboard.html";
    else window.location.href="patient-dashboard.html";
  } else {
    alert("Invalid OTP!");
  }
}

// FORGOT PASSWORD (demo)
function forgotPassword(){
  const email = document.getElementById("forgotEmail").value.trim();
  const user = users.find(u=>u.email===email);
  if(!user){ alert("Email not found!"); return; }
  const newPass = prompt("Enter new password:");
  if(newPass){ user.password = newPass; alert("Password updated! Login now."); showForm('login-section'); }
}
</script>
