<script>
// Demo in-memory users
let users = [
  {name:"Admin", email:"admin@gmail.com", password:"admin123", role:"admin"},
  {name:"Doctor", email:"doctor@gmail.com", password:"doc123", role:"doctor"},
  {name:"Receptionist", email:"reception@gmail.com", password:"rec123", role:"receptionist"}
];
let tempUser = null;

// Show specific form
function showForm(id){
  document.querySelectorAll('.auth-section').forEach(s=>s.style.display='none');
  document.getElementById(id).style.display='block';
}

// Register
function register(){
  const name=document.getElementById("registerName").value;
  const email=document.getElementById("registerEmail").value;
  const password=document.getElementById("registerPassword").value;
  const confirm=document.getElementById("registerConfirmPassword").value;
  const role=document.getElementById("registerRole").value;
  if(!name||!email||!password||!confirm||!role){ alert("Fill all fields"); return; }
  if(password!==confirm){ alert("Passwords do not match"); return; }
  if(users.find(u=>u.email===email)){ alert("Email already exists"); return; }
  users.push({name,email,password,role});
  alert("Signup successful!"); showForm("login-form");
}

// Login
function login(){
  const email=document.getElementById("loginEmail").value;
  const password=document.getElementById("loginPassword").value;
  const user = users.find(u=>u.email===email && u.password===password);
  if(user){ tempUser=user; showForm('twofa-form'); }
  else alert("Invalid credentials");
}

// Verify 2FA and redirect to index.html
function verify2FA(){
  const code=document.getElementById("twofaCode").value;
  if(code==="123456"){ // demo OTP
    sessionStorage.setItem("loggedIn","true");
    sessionStorage.setItem("role", tempUser.role);
    sessionStorage.setItem("name", tempUser.name);

    // Redirect to index.html
    window.location.href = "index.html";
  } else { alert("Invalid OTP"); }
}

// Forgot Password
function forgotPassword(){
  const email=document.getElementById("forgotEmail").value;
  if(users.find(u=>u.email===email)){ alert("Reset link sent (demo)"); }
  else alert("Email not found");
}

// Sidebar toggle
document.getElementById('toggleSidebar').addEventListener('click', ()=>{
  document.getElementById('sidebar').classList.toggle('open');
  document.getElementById('main').classList.toggle('shifted');
});

// Sidebar content display
function showContent(section){
  const area = document.getElementById('content-area');
  switch(section){
    case 'appointments': 
      area.innerHTML=`<h3>Appointments</h3><p>List of appointments will be displayed here.</p>`; break;
    case 'patients': 
      area.innerHTML=`<h3>Patients</h3><p>Patient records will be displayed here.</p>`; break;
    case 'billing': 
      area.innerHTML=`<h3>Billing</h3><p>Billing info will be displayed here.</p>`; break;
    case 'reports': 
      area.innerHTML=`<h3>Reports</h3><p>Reports will be displayed here.</p>`; break;
    case 'settings': 
      area.innerHTML=`<h3>Settings</h3><p>Profile and settings can be changed here.</p>`; break;
  }
}

// Logout
function logout(){
  sessionStorage.clear();
  alert("Logged out!");
  location.reload();
}
